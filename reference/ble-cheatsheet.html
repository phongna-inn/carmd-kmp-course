Callbacks into a Kotlin Flow

// Android: the adapter is the callbackFlow
override fun scan(): Flow<Device> = callbackFlow {
    val callback = object : ScanCallback() {
        override fun onScanResult(type: Int, r: ScanResult) { trySend(r.toDevice()) }
        override fun onScanFailed(code: Int) { close(ScanException("…$code")) }
    }
    scanner.startScan(null, settings, callback)
    awaitClose {                                   // mandatory: runs on cancel
        try {
            scanner.stopScan(callback)             // even when adapter.isEnabled is false
        } catch (e: IllegalStateException) {
            // BT fully off, and going fully off drops every scan
        }
    }
}

// iOS: Swift can't build a Flow (SKIE converts Flow → AsyncSequence only),
// so Kotlin in iosMain builds it and Swift feeds it
fun scanFlow(radio: IosRadio): Flow<Device> = callbackFlow {
    radio.startScan(ScanFeed(this))                // Swift calls feed.heard / feed.failed
    awaitClose { radio.stopScan() }
}
// Swift: func scan() -> SkieSwiftFlow<Device> { scanFlow(radio: self) }

A connection as a Flow

// Port: collecting connects, each value is the link's state, cancelling disconnects
interface BleConnection { fun connect(device: Device): Flow<ConnectionState> }

// Android adapter
override fun connect(device: Device) = callbackFlow {
    val callback = object : BluetoothGattCallback() {
        override fun onConnectionStateChange(g: BluetoothGatt, status: Int, state: Int) {
            if (status == GATT_SUCCESS && state == STATE_CONNECTED) trySend(Connected(device))
            else close(ConnectionException("…$status"))
        }
    }
    send(Connecting(device))
    val gatt = adapter.getRemoteDevice(device.id).connectGatt(context, false, callback, TRANSPORT_LE)
    awaitClose { gatt.disconnect(); gatt.close() }   // every way out closes the client
}

// iOS: Kotlin builds the Flow (ConnectFlow.kt in iosMain), Swift feeds it
fun connectFlow(link: IosLink, device: Device) = callbackFlow {
    link.open(device, LinkFeed(this))                // Swift: feed.changed / feed.failed
    awaitClose { link.close() }                      // Swift: cancelPeripheralConnection
}

// Service: the connection lasts as long as the caller's coroutine
override suspend fun connect(device: Device) {
    /* permission, one at a time */ state.value = Connecting(device)
    try {
        coroutineScope {
            link = launch {                          // disconnect() = link?.cancel(): returns normally
                val giveUp = launch { delay(10.seconds); throw ConnectionException("…") }
                connection.connect(device).collect { state.value = it; if (it is Connected) giveUp.cancel() }
            }
        }
    } finally { state.value = Disconnected }
}
// ViewModel: viewModelScope.launch { session.connect(device) }: survives rotation, ends when cleared

Requests as suspend funs

// Port: one request at a time on the open link
suspend fun discoverServices(): List<DeviceService>
suspend fun read(characteristic: Characteristic): ByteArray

// Android: the callback completes the waiting request
private suspend fun request(send: () -> Boolean): Answer {
    val pending = CompletableDeferred<Answer>(); answer = pending
    try { if (!send()) throw ConnectionException("…"); return pending.await() } finally { answer = null }
}
// in awaitClose: answer?.completeExceptionally(ConnectionException("The link closed …"))

// iOS: the Kotlin half implements the port; Swift answers through a reply
override suspend fun read(c: Characteristic): ByteArray =
    suspendCancellableCoroutine { link.read(c, ValueReply(it)) }   // Swift: reply.read(value:) / reply.failed(message:)
// A Swift error thrown into Kotlin arrives as ObjCErrorException, not ConnectionException

// Service: one request at a time, only what can be read, decoded in the app core
private val requests = Mutex()
if (CharacteristicProperty.Read !in c.properties) throw ConnectionException("…")
val bytes = requests.withLock { connection.read(c) }
return ValueDecoder.decode(c.uuid, bytes)

A subscription as a Flow

// Port: switches are requests, values are a stream (hot: collect first, then subscribe)
suspend fun subscribe(characteristic: Characteristic)
suspend fun unsubscribe(characteristic: Characteristic)
fun notifications(characteristic: Characteristic): Flow<ByteArray>
// adapters: a MutableSharedFlow, tryEmit from the callback (thread-safe, doesn't suspend)

// Service: collecting subscribes, cancelling unsubscribes, the link's end ends it
override fun subscribe(c: Characteristic): Flow<String> = channelFlow {
    val values = launch(start = CoroutineStart.UNDISPATCHED) {      // listening before asking
        connection.notifications(c).collect { send(ValueDecoder.decode(c.uuid, it)) }
    }
    try {
        requests.withLock { connection.subscribe(c) }
        current.join()                                               // the link's Job
    } catch (e: CancellationException) {                             // the collector went away
        if (current.isActive) withContext(NonCancellable) { stopNotifications(c) }
        throw e
    }
    values.cancel()                                                  // a SharedFlow never completes
}

Traps

  • Simulator and emulator. The iOS Simulator has no BLE radio (.unsupported). The Android emulator's Bluetooth is simulated: it never hears your Mac. Real peripherals need real phones.
  • The iOS delegate is weak. Keep the delegate alive (here: Koin holds the adapter, which is its own delegate), or callbacks go nowhere.
  • Android scan throttling. Too many scan starts in a short time and the next one silently returns nothing; logcat says "scanning too frequently". We hit it with six starts in about ten seconds.
  • Check the permission before the radio. On iOS, creating the manager is a first use of Bluetooth and can show the prompt.
  • Always time-limit a scan. "Never scan on a loop, and always set a time limit on your scan."
  • A connection kept in the screen. A BluetoothGatt in an Activity or in Compose state is lost on rotation while Android keeps its client registered; a second Connect registers another. Keep it in the adapter, and run the connection in viewModelScope, not rememberCoroutineScope().
  • An unretained peripheral. CoreBluetooth cancels a pending connection when its CBPeripheral is deallocated.
  • Disappear is not "gone". onDisappear / viewDidDisappear also run when a screen is pushed on top. End a connection when the ViewModel is cleared, not there.
  • iOS connection attempts never time out. The app sets its own limit (10 s, in the service).
  • Always close() a BluetoothGatt when done: put it in awaitClose, so every path runs it.
  • The peripheral's delegate (iOS). Separate from the manager's, weak, one per peripheral, no required methods: unset or released, discovery and reads are answered to no one and nothing fails. Set it as soon as you have the peripheral.
  • One GATT request at a time (Android). readCharacteristic returns false while another read or write waits for its callback (discoverServices() doesn't count). Queue them (here: a Mutex in the service).
  • A signed Byte. Without and 0xFF, 200 reads as −56.
  • Byte order. GATT values are little endian unless the characteristic says otherwise.
  • A request still waiting when the adapter closes the link never gets its callback (Android's close() unregisters it): fail the request in the same close.
  • Notifications need both switches (Android). setCharacteristicNotification alone asks the device for nothing; the CCCD write alone isn't passed to the app. Wait for onDescriptorWrite before the next request.
  • A hot flow keeps nothing for later. A value emitted into a SharedFlow with no subscriber and no replay is lost: start collecting before switching notifications on.
  • Suspending cleanup in a cancelled coroutine throws CancellationException at the first real suspension: wrap it in withContext(NonCancellable).
  • A read and a notification look the same on iOS. Both arrive in didUpdateValueFor; a value that answers a waiting read doesn't also reach the subscription.
  • Views rebuilt on every value swallow taps (XML Views, UIKit): the pressed button is gone by the release. Rebuild rows only when they change; update values in place.
